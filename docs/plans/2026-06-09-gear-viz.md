# Honest Gear Visualization Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Show one drive's gear structure honestly — an RPM-vs-speed scatter with detected gear-ratio lines + a "gears used this drive" breakdown (relative labels, no absolute gear numbers), in the advanced-graphs view.

**Architecture:** Backend derives gear ratios on the existing same-instant `resample_1hz` grid (avoiding the independently-downsampled-signal alignment trap), assigns each moving sample to a gear (or "unassigned" for clutch/shift), and emits a pre-paired scatter in a new `Gears` model field. The frontend `GearChart` only draws it. Per-drive, stateless. Spec: `docs/specs/2026-06-09-gear-viz-design.md`.

**Tech Stack:** Python (FastAPI/pandas/numpy, pytest) for analysis; React + TypeScript + Recharts + Vitest for frontend. Backend commands from `analysis/` (use `.venv/bin/python`); frontend from `frontend/`.

```python
import numpy as np
import pandas as pd

from app import metrics
from app.models import GearPoint, GearUsage, Gears

_MIN_KMH = 6.0          # below this is clutch-slip / creep, not a clean gear
_MIN_RPM = 900.0        # above idle
_MAX_SCATTER = 250      # downsample cap for the chart
_ASSIGN_TOL = 0.10      # a sample within 10% of a ratio is "in" that gear
_MERGE_FRAC = 0.06      # merge histogram peaks within 6% of the ratio span
_LABELS = ("tallest used", "one below", "two below", "three below", "four below")


def _label(i: int) -> str:
    return _LABELS[i] if i < len(_LABELS) else f"{i} below"


def _moving(df: pd.DataFrame) -> pd.DataFrame | None:
    """1 Hz same-instant grid of moving samples with their RPM/speed ratio."""
    src = metrics.speed_source(df)
    if src is None:
        return None
    grid = metrics.resample_1hz(df, [src, "Engine RPM"]).dropna(subset=[src, "Engine RPM"])
    g = grid[(grid[src] >= _MIN_KMH) & (grid["Engine RPM"] >= _MIN_RPM)].copy()
    if g.empty:
        return None
    g = g.rename(columns={src: "speed", "Engine RPM": "rpm"})
    g["ratio"] = g["rpm"] / g["speed"]
    return g[["speed", "rpm", "ratio"]].reset_index(drop=True)


def _detect_ratios(g: pd.DataFrame) -> list[float]:
    """Histogram-peak the steady-sample ratios -> gear ratios ascending (tallest first)."""
    dspeed = g["speed"].diff().abs()
    dratio = g["ratio"].diff().abs()
    steady = g["ratio"][(dspeed <= 1.5) & (dratio <= g["ratio"] * 0.04)].to_numpy()
    if len(steady) < 40:
        return []
    lo, hi = float(np.percentile(steady, 1)), float(np.percentile(steady, 99))
    if hi <= lo:
        return []
    pad = (hi - lo) * 0.04  # so a peak sitting at lo/hi lands in an interior bin
    hist, edges = np.histogram(steady, bins=np.linspace(lo - pad, hi + pad, 62))
    centers = (edges[:-1] + edges[1:]) / 2
    thr = max(hist.max() * 0.15, len(steady) * 0.01)
    span = hi - lo
    peaks: list[float] = []
    heights: list[int] = []
    for i in range(1, len(hist) - 1):
        if hist[i] >= thr and hist[i] >= hist[i - 1] and hist[i] >= hist[i + 1]:
            if peaks and centers[i] - peaks[-1] <= span * _MERGE_FRAC:
                if hist[i] > heights[-1]:
                    peaks[-1], heights[-1] = float(centers[i]), int(hist[i])
            else:
                peaks.append(float(centers[i]))
                heights.append(int(hist[i]))
    return [round(p, 1) for p in sorted(peaks)]


def _assign(ratio: float, ratios: list[float]) -> int | None:
    if not ratios:
        return None
    rel = [abs(ratio - r) / r for r in ratios]
    j = int(np.argmin(rel))
    return j if rel[j] <= _ASSIGN_TOL else None


def build_gears(df: pd.DataFrame) -> Gears:
    g = _moving(df)
    if g is None or g.empty:
        return Gears()
    ratios = _detect_ratios(g)
    assigned = [_assign(float(r), ratios) for r in g["ratio"]]
    n = len(g)
    usage = [
        GearUsage(label=_label(i), ratio=r,
                  pct=round(sum(1 for a in assigned if a == i) / n * 100.0, 1))
        for i, r in enumerate(ratios)
    ]
    unassigned = sum(1 for a in assigned if a is None)
    step = max(1, int(np.ceil(n / _MAX_SCATTER)))
    points = [
        GearPoint(speedKmh=round(float(g["speed"].iloc[k]), 1),
                  rpm=round(float(g["rpm"].iloc[k]), 0),
                  gear=assigned[k])
        for k in range(0, n, step)
    ]
    return Gears(ratios=ratios, usage=usage,
                 unassignedPct=round(unassigned / n * 100.0, 1), points=points)
```

```tsx
import { CartesianGrid, ReferenceLine, ResponsiveContainer, Scatter, ScatterChart, Tooltip, XAxis, YAxis, ZAxis } from 'recharts';
import type { Gears } from '../types';
import ChartTable from './ChartTable';

const GEAR_COLORS = ['#0072B2', '#E69F00', '#009E73', '#CC79A7', '#56B4E9']; // Okabe–Ito
const UNASSIGNED = '#9aa0a6';

interface Props {
  gears?: Gears;
}

export default function GearChart({ gears }: Props) {
  if (!gears || gears.points.length === 0) return null;
  const { ratios, usage, unassignedPct, points } = gears;
  const xMax = Math.max(...points.map((p) => p.speedKmh), 1);
  const byGear = new Map<string, typeof points>();
  for (const p of points) {
    const k = p.gear == null ? 'u' : String(p.gear);
    (byGear.get(k) ?? byGear.set(k, []).get(k)!).push(p);
  }
  const summary = ratios.length > 0
    ? `RPM vs speed scatter: ${ratios.length} gears detected this drive (relative, not absolute), ${points.length} points`
    : `RPM vs speed scatter: gears could not be separated for this drive, ${points.length} points`;
  return (
    <section className="card">
      <h3 className="section-title">Gears <small>· this drive only · relative, not absolute gear numbers</small></h3>
      <div style={{ width: '100%', height: 260 }} role="img" aria-label={summary}>
        <ResponsiveContainer>
          <ScatterChart margin={{ top: 8, right: 16, bottom: 4, left: 0 }}>
            <CartesianGrid strokeDasharray="3 3" />
            <XAxis type="number" dataKey="speedKmh" name="Speed" unit=" km/h" />
            <YAxis type="number" dataKey="rpm" name="RPM" width={48} />
            <ZAxis range={[18, 18]} />
            <Tooltip cursor={{ strokeDasharray: '3 3' }} />
            {ratios.map((r, i) => (
              <ReferenceLine key={`r${i}`} ifOverflow="extendDomain"
                stroke={GEAR_COLORS[i % GEAR_COLORS.length]} strokeDasharray="4 4"
                segment={[{ x: 0, y: 0 }, { x: xMax, y: r * xMax }]} />
            ))}
            {[...byGear.entries()].map(([k, pts]) => (
              <Scatter key={k} data={pts}
                name={k === 'u' ? 'clutch/shift' : (usage[Number(k)]?.label ?? `gear ${k}`)}
                fill={k === 'u' ? UNASSIGNED : GEAR_COLORS[Number(k) % GEAR_COLORS.length]}
                isAnimationActive={false} />
            ))}
          </ScatterChart>
        </ResponsiveContainer>
      </div>
      <ul className="gear-usage">
        {usage.map((u) => (
          <li key={u.label}>{u.label}: {u.pct}% <small>(ratio {u.ratio})</small></li>
        ))}
        {unassignedPct > 0 && <li>clutch/shift/coast: {unassignedPct}%</li>}
        {ratios.length === 0 && <li>Gears couldn’t be separated for this drive.</li>}
      </ul>
      <ChartTable caption={summary} columns={['Gear', 'Ratio (rpm/km·h⁻¹)', '% of moving time']}
        rows={[
          ...usage.map((u) => [u.label, u.ratio, u.pct]),
          ...(unassignedPct > 0 ? [['clutch/shift/coast', '—', unassignedPct]] : []),
        ]} />
    </section>
  );
}
```
